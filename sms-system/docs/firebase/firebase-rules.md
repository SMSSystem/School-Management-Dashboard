# Firebase Rules (As Seen In Firebase Console)

> **Reference only.** The authoritative copy of these rules lives in the project's Firebase Console, not in this repository. This file exists for documentation and review purposes. It must only be updated to reflect rules that have already been deployed to the Firebase Console — do not edit this file to propose or stage rule changes.

## Architecture (post-migration)

All institution data now lives under `/institutions/{institutionId}/` subcollections. The only top-level collections are `/users` (global identity) and `/institutions` (institution records).

The old top-level `teachers/` and `students/` collections are eliminated — per-institution role data now lives in `/institutions/{institutionId}/members/{userId}`.

> **Composite indexes required (institution subcollection scope):**
>
> - **`generalAttendance`** — Collection scope, fields `classId ASC · date ASC · session ASC`. Required for the live register query and admin overdue detection.
> - **`subjectAttendance`** — Collection scope, fields `subjectId ASC · classId ASC · sessionDate ASC`. Required for the subject register week query.
> - **`results`** — Collection scope, fields `studentId ASC · termId ASC`.
> - **`reportCards`** — Collection scope, fields `classId ASC · termId ASC`.
> - **`members`** — Collection scope, fields `role ASC · classId ASC`. Required for student roster queries.
> - **`members`** — Collection scope, fields `role ASC · status ASC`.

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helpers ────────────────────────────────────────────────────────────
    function isSignedIn() {
      return request.auth != null;
    }

    // Primary profile: role and institutionId are kept on users/{uid} for fast
    // auth and security rule evaluation.
    function me() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function myRole() {
      return me().role;
    }

    function myInstitutionId() {
      return me().institutionId;
    }

    // Institution membership: school-specific fields (departmentId, memberType, etc.)
    function myMember(institutionId) {
      return get(/databases/$(database)/documents/institutions/$(institutionId)/members/$(request.auth.uid)).data;
    }

    function isSuperAdmin() {
      return isSignedIn() && myRole() == 'super_admin';
    }

    function isAdmin() {
      return isSignedIn() && myRole() == 'institution_admin';
    }

    function isAdminOrAbove() {
      return isSuperAdmin() || isAdmin();
    }

    function isTeacher() {
      return isSignedIn() && (myRole() == 'senior_teacher' || myRole() == 'regular_teacher');
    }

    function isSeniorTeacher() {
      return isSignedIn() && myRole() == 'senior_teacher';
    }

    function isTeacherOrAbove() {
      return isAdminOrAbove() || isTeacher();
    }

    function isParent() {
      return isSignedIn() && myRole() == 'parent';
    }

    function isOwner(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    function roleNotChanged() {
      return !('role' in request.resource.data.diff(resource.data).affectedKeys());
    }

    function institutionNotChanged() {
      return !('institutionId' in request.resource.data.diff(resource.data).affectedKeys());
    }

    // Super admin bypasses institution scoping ('*' sentinel on users/{uid})
    function sameInstitution(institutionId) {
      return isSuperAdmin() || myInstitutionId() == institutionId;
    }

    // For creates: verify the caller belongs to the target institution
    function writingToMyInstitution(institutionId) {
      return isSuperAdmin() || myInstitutionId() == institutionId;
    }

    // Class teacher check — reads from the institution's classes subcollection
    function isClassTeacherFor(institutionId, docClassId) {
      return isTeacher()
        && get(/databases/$(database)/documents/institutions/$(institutionId)/classes/$(docClassId)).data.classTeacherId
           == request.auth.uid;
    }

    // Senior teacher department check — reads from members subcollection
    function isSeniorTeacherFor(institutionId, docDepartmentId) {
      let member = myMember(institutionId);
      return myRole() == 'senior_teacher'
        && member.departmentId == docDepartmentId;
    }

    // ── Users ──────────────────────────────────────────────────────────────
    // Global identity. role and institutionId stay here for security rule
    // performance. School-specific data lives in institutions/{id}/members/{uid}.
    match /users/{uid} {
      allow read: if isOwner(uid)
        || (isAdminOrAbove() && sameInstitution(resource.data.institutionId))
        || (isTeacher() && sameInstitution(resource.data.institutionId) && resource.data.role == 'student');

      allow create: if isSuperAdmin()
        || (isAdmin() && request.resource.data.institutionId == myInstitutionId()
            && request.resource.data.role in ['senior_teacher', 'regular_teacher', 'student', 'parent']);

      allow update: if (isOwner(uid) && roleNotChanged() && institutionNotChanged())
        || (isAdminOrAbove() && sameInstitution(resource.data.institutionId)
            && institutionNotChanged() && roleNotChanged());

      allow delete: if isSuperAdmin();
    }

    // ── Activity log (per-user subcollection) ─────────────────────────────
    match /users/{uid}/activity_log/{eventId} {
      allow read:   if isOwner(uid);
      allow create: if isOwner(uid)
        && request.resource.data.uid == uid
        && request.resource.data.keys().hasAll(['eventType','detail','timestamp','uid','institutionId']);
      allow update, delete: if false;
    }

    // ── Institutions ───────────────────────────────────────────────────────
    match /institutions/{institutionId} {
      allow read: if isSuperAdmin() || myInstitutionId() == institutionId;
      allow create: if isSuperAdmin();
      allow update: if isSuperAdmin()
        || (isAdmin() && myInstitutionId() == institutionId);
      allow delete: if isSuperAdmin();
    }

    // ── Audit log (institution-scoped subcollection) ───────────────────────
    match /institutions/{institutionId}/audit_log/{eventId} {
      allow read: if isSuperAdmin()
        || (isAdmin() && myInstitutionId() == institutionId);
      allow create: if (isSuperAdmin() || (isAdmin() && myInstitutionId() == institutionId))
        && request.resource.data.institutionId == institutionId
        && request.resource.data.keys().hasAll([
             'eventType','detail','targetUid','targetName',
             'performedBy','performedByName','timestamp','institutionId'
           ]);
      allow update, delete: if false;
    }

    // ── Members ────────────────────────────────────────────────────────────
    // Replaces the old top-level teachers/{uid} and students/{uid} collections.
    // One document per user per institution; userId == doc ID.
    match /institutions/{institutionId}/members/{userId} {
      allow read: if isSuperAdmin()
        || (isSignedIn() && myInstitutionId() == institutionId && isAdminOrAbove())
        || (isSignedIn() && myInstitutionId() == institutionId && isTeacher() && resource.data.role == 'student')
        || isOwner(userId);
      allow create: if isSuperAdmin()
        || (isAdmin() && myInstitutionId() == institutionId);
      allow update: if isSuperAdmin()
        || (isAdmin() && myInstitutionId() == institutionId)
        || (isOwner(userId) && !('role' in request.resource.data.diff(resource.data).affectedKeys()));
      allow delete: if isSuperAdmin()
        || (isAdmin() && myInstitutionId() == institutionId);
    }

    // ── Classes ────────────────────────────────────────────────────────────
    match /institutions/{institutionId}/classes/{classId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Subjects ───────────────────────────────────────────────────────────
    match /institutions/{institutionId}/subjects/{subjectId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Terms ──────────────────────────────────────────────────────────────
    match /institutions/{institutionId}/terms/{termId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Academic Years ─────────────────────────────────────────────────────
    match /institutions/{institutionId}/academicYears/{yearId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create, update, delete: if isSignedIn()
        && isAdmin()
        && myInstitutionId() == institutionId;
    }

    // ── Non-School Days ────────────────────────────────────────────────────
    match /institutions/{institutionId}/nonSchoolDays/{dayId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create, update, delete: if isSignedIn()
        && isAdmin()
        && myInstitutionId() == institutionId;
    }

    // ── Houses ─────────────────────────────────────────────────────────────
    match /institutions/{institutionId}/houses/{houseId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdmin() && writingToMyInstitution(institutionId);
      allow update: if isAdmin() && myInstitutionId() == institutionId;
      allow delete: if isAdmin() && myInstitutionId() == institutionId;
    }

    // ── Departments ────────────────────────────────────────────────────────
    match /institutions/{institutionId}/departments/{departmentId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Student-Parent Relationships ───────────────────────────────────────
    // Doc ID: {parentId}_{studentId}
    match /institutions/{institutionId}/studentParents/{docId} {
      allow read: if (isTeacherOrAbove() && myInstitutionId() == institutionId)
        || resource.data.studentId == request.auth.uid
        || resource.data.parentId == request.auth.uid;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Results ────────────────────────────────────────────────────────────
    match /institutions/{institutionId}/results/{resultId} {
      allow read: if (isTeacherOrAbove() && myInstitutionId() == institutionId)
        || resource.data.studentId == request.auth.uid
        || (isParent() && exists(/databases/$(database)/documents/institutions/$(institutionId)/studentParents/$(request.auth.uid + '_' + resource.data.studentId)));

      allow create: if isTeacherOrAbove()
        && writingToMyInstitution(institutionId)
        && request.resource.data.teacherId == request.auth.uid
        && request.resource.data.score <= request.resource.data.maxScore
        && (isAdminOrAbove()
          || isSeniorTeacherFor(institutionId, request.resource.data.departmentId)
          || (myRole() == 'regular_teacher'
              && request.auth.uid in get(
                   /databases/$(database)/documents/institutions/$(institutionId)/subjects/$(request.resource.data.subjectId)
                 ).data.teacherIds));

      allow update: if myInstitutionId() == institutionId
        && request.resource.data.score <= request.resource.data.maxScore
        && (isAdminOrAbove()
          || isSeniorTeacherFor(institutionId, resource.data.departmentId)
          || (myRole() == 'regular_teacher'
              && resource.data.teacherId == request.auth.uid
              && request.auth.uid in get(
                   /databases/$(database)/documents/institutions/$(institutionId)/subjects/$(request.resource.data.subjectId)
                 ).data.teacherIds));

      allow delete: if (isAdminOrAbove() && myInstitutionId() == institutionId)
        || (isTeacher()
            && myInstitutionId() == institutionId
            && resource.data.source == 'gradebook');
    }

    // ── Feedback Comments ──────────────────────────────────────────────────
    match /institutions/{institutionId}/feedbackComments/{docId} {
      allow read: if (isTeacherOrAbove() && myInstitutionId() == institutionId)
        || resource.data.studentId == request.auth.uid
        || (isParent() && exists(/databases/$(database)/documents/institutions/$(institutionId)/studentParents/$(request.auth.uid + '_' + resource.data.studentId)));

      allow create: if writingToMyInstitution(institutionId)
        && (isAdminOrAbove()
          || (request.resource.data.teacherId == request.auth.uid
              && (isSeniorTeacherFor(institutionId, request.resource.data.departmentId)
                  || (myRole() == 'regular_teacher'
                      && request.auth.uid in get(
                           /databases/$(database)/documents/institutions/$(institutionId)/subjects/$(request.resource.data.subjectId)
                         ).data.teacherIds))));

      allow update: if myInstitutionId() == institutionId
        && (isAdminOrAbove()
          || isSeniorTeacherFor(institutionId, resource.data.departmentId)
          || (myRole() == 'regular_teacher'
              && resource.data.teacherId == request.auth.uid
              && request.auth.uid in get(
                   /databases/$(database)/documents/institutions/$(institutionId)/subjects/$(request.resource.data.subjectId)
                 ).data.teacherIds));

      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Student Activities ────────────────────────────────────────────────
    match /institutions/{institutionId}/studentActivities/{id} {
      allow read: if isSignedIn()
        && myInstitutionId() == institutionId
        && (isTeacherOrAbove()
          || resource.data.studentId == request.auth.uid
          || (isParent() && exists(/databases/$(database)/documents/institutions/$(institutionId)/studentParents/$(request.auth.uid + '_' + resource.data.studentId))));
      allow create: if isAdmin() && writingToMyInstitution(institutionId);
      allow update: if isAdmin() && myInstitutionId() == institutionId;
      allow delete: if isAdmin() && myInstitutionId() == institutionId;
    }

    // ── Student Responsibilities ──────────────────────────────────────────
    match /institutions/{institutionId}/studentResponsibilities/{id} {
      allow read: if isSignedIn()
        && myInstitutionId() == institutionId
        && (isTeacherOrAbove()
          || resource.data.studentId == request.auth.uid
          || (isParent() && exists(/databases/$(database)/documents/institutions/$(institutionId)/studentParents/$(request.auth.uid + '_' + resource.data.studentId))));
      allow create: if isAdmin() && writingToMyInstitution(institutionId);
      allow update: if isAdmin() && myInstitutionId() == institutionId;
      allow delete: if isAdmin() && myInstitutionId() == institutionId;
    }

    // ── Report Card Comments ──────────────────────────────────────────────
    match /institutions/{institutionId}/reportCardComments/{id} {
      allow read: if isSignedIn()
        && myInstitutionId() == institutionId
        && isAdminOrAbove();
      allow create: if isAdmin() && writingToMyInstitution(institutionId);
      allow update: if isAdmin() && myInstitutionId() == institutionId;
      allow delete: if isAdmin() && myInstitutionId() == institutionId;
    }

    // ── Report Cards ──────────────────────────────────────────────────────
    match /institutions/{institutionId}/reportCards/{id} {
      allow read: if isSignedIn()
        && myInstitutionId() == institutionId
        && (isTeacherOrAbove()
          || resource.data.studentId == request.auth.uid
          || (isParent() && exists(/databases/$(database)/documents/institutions/$(institutionId)/studentParents/$(request.auth.uid + '_' + resource.data.studentId))));
      allow create: if isAdmin() && writingToMyInstitution(institutionId);
      allow update: if isAdmin() && myInstitutionId() == institutionId;
      allow delete: if isAdmin() && myInstitutionId() == institutionId;
    }

    // ── Reports (legacy — locked) ─────────────────────────────────────────
    match /institutions/{institutionId}/reports/{docId} {
      allow read, write: if false;
    }

    // ── General Attendance Register ────────────────────────────────────────
    // Composite index required: classId ASC · date ASC · session ASC
    match /institutions/{institutionId}/generalAttendance/{docId} {
      allow read: if isSignedIn()
        && myInstitutionId() == institutionId
        && (
          isAdminOrAbove()
          || isSeniorTeacher()
          || (
            myRole() == 'student'
            && get(/databases/$(database)/documents/users/$(request.auth.uid))
                 .data.classId == resource.data.classId
          )
          || isParent()
        );

      allow create, update: if isSignedIn()
        && writingToMyInstitution(institutionId)
        && (
          isAdmin()
          || (
            isSeniorTeacher()
            && get(/databases/$(database)/documents/users/$(request.auth.uid))
                 .data.assignedClassId == request.resource.data.classId
          )
        );

      allow delete: if isSignedIn()
        && isAdmin()
        && myInstitutionId() == institutionId;
    }

    // ── Subject Enrollments ────────────────────────────────────────────────
    // Doc ID: {subjectId}_{classId}
    match /institutions/{institutionId}/subjectEnrollments/{enrollmentId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Subject Attendance Register ────────────────────────────────────────
    // Composite index required: subjectId ASC · classId ASC · sessionDate ASC
    match /institutions/{institutionId}/subjectAttendance/{docId} {
      allow read: if isSignedIn()
        && myInstitutionId() == institutionId
        && (
          isAdminOrAbove()
          || (
            myRole() == 'regular_teacher'
            && request.auth.uid in get(
                 /databases/$(database)/documents/institutions/$(institutionId)/subjects/$(resource.data.subjectId)
               ).data.teacherIds
          )
          || myRole() == 'student'
          || isParent()
        );

      allow create, update: if isSignedIn()
        && writingToMyInstitution(institutionId)
        && (
          isAdmin()
          || (
            myRole() == 'regular_teacher'
            && request.auth.uid == request.resource.data.teacherId
            && request.auth.uid in get(
                 /databases/$(database)/documents/institutions/$(institutionId)/subjects/$(request.resource.data.subjectId)
               ).data.teacherIds
          )
        );

      allow delete: if isSignedIn()
        && isAdmin()
        && myInstitutionId() == institutionId;
    }

    // ── Attendance Summaries ───────────────────────────────────────────────
    match /institutions/{institutionId}/attendanceSummaries/{id} {
      allow read: if isSignedIn()
        && ((resource == null && isTeacherOrAbove() && myInstitutionId() == institutionId)
            || (resource != null
                && myInstitutionId() == institutionId
                && (isTeacherOrAbove()
                  || resource.data.studentId == request.auth.uid
                  || (isParent()
                      && exists(/databases/$(database)/documents/institutions/$(institutionId)/studentParents/$(request.auth.uid + '_' + resource.data.studentId))))));
      allow create, update: if isSignedIn()
        && writingToMyInstitution(institutionId)
        && (isAdmin() || isSeniorTeacher());
      allow delete: if isAdmin() && myInstitutionId() == institutionId;
    }

    // ── Timetable Slots ────────────────────────────────────────────────────
    match /institutions/{institutionId}/timetableSlots/{slotId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;

      allow create: if (isAdminOrAbove() && myInstitutionId() == institutionId)
        || (isSeniorTeacher()
            && myInstitutionId() == institutionId
            && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.canGenerateSchedule == true);

      allow update, delete: if (isAdminOrAbove() && myInstitutionId() == institutionId)
        || (isSeniorTeacher()
            && myInstitutionId() == institutionId
            && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.canGenerateSchedule == true);
    }

    // ── Gradebooks ─────────────────────────────────────────────────────────
    match /institutions/{institutionId}/gradebooks/{gradebookId} {
      allow read: if isTeacherOrAbove() && myInstitutionId() == institutionId;
      allow create: if isTeacherOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isTeacherOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;

      match /columns/{colId} {
        allow read: if isTeacherOrAbove() && myInstitutionId() == institutionId;
        allow create: if isTeacherOrAbove() && writingToMyInstitution(institutionId);
        allow update: if isTeacherOrAbove() && myInstitutionId() == institutionId;
        allow delete: if isSignedIn()
          && myInstitutionId() == institutionId
          && (isAdmin() || request.auth.uid == resource.data.createdBy);
      }
    }

    // ── Events ─────────────────────────────────────────────────────────────
    match /institutions/{institutionId}/events/{eventId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Announcements ──────────────────────────────────────────────────────
    match /institutions/{institutionId}/announcements/{announcementId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isAdminOrAbove() && writingToMyInstitution(institutionId);
      allow update: if isAdminOrAbove() && myInstitutionId() == institutionId;
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Lessons ────────────────────────────────────────────────────────────
    match /institutions/{institutionId}/lessons/{lessonId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isTeacherOrAbove() && writingToMyInstitution(institutionId);
      allow update: if myInstitutionId() == institutionId
        && (isAdminOrAbove()
          || (isTeacher() && resource.data.teacherId == request.auth.uid)
          || isSeniorTeacherFor(institutionId, resource.data.departmentId));
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Exams ──────────────────────────────────────────────────────────────
    match /institutions/{institutionId}/exams/{examId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isTeacherOrAbove() && writingToMyInstitution(institutionId);
      allow update: if myInstitutionId() == institutionId
        && (isAdminOrAbove()
          || (isTeacher() && resource.data.teacherId == request.auth.uid)
          || isSeniorTeacherFor(institutionId, resource.data.departmentId));
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Assignments ────────────────────────────────────────────────────────
    match /institutions/{institutionId}/assignments/{assignmentId} {
      allow read: if isSignedIn() && myInstitutionId() == institutionId;
      allow create: if isTeacherOrAbove() && writingToMyInstitution(institutionId);
      allow update: if myInstitutionId() == institutionId
        && (isAdminOrAbove()
          || (isTeacher() && resource.data.teacherId == request.auth.uid)
          || isSeniorTeacherFor(institutionId, resource.data.departmentId));
      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Attendance (legacy per-student records) ────────────────────────────
    match /institutions/{institutionId}/attendance/{recordId} {
      allow read: if (isTeacherOrAbove() && myInstitutionId() == institutionId)
        || isOwner(resource.data.studentId)
        || (isParent() && exists(/databases/$(database)/documents/institutions/$(institutionId)/studentParents/$(request.auth.uid + '_' + resource.data.studentId)));

      allow create: if writingToMyInstitution(institutionId)
        && (isAdminOrAbove()
          || isClassTeacherFor(institutionId, request.resource.data.classId)
          || isSeniorTeacherFor(institutionId, request.resource.data.departmentId));

      allow update: if myInstitutionId() == institutionId
        && (isAdminOrAbove()
          || isClassTeacherFor(institutionId, resource.data.classId)
          || isSeniorTeacherFor(institutionId, resource.data.departmentId));

      allow delete: if isAdminOrAbove() && myInstitutionId() == institutionId;
    }

    // ── Collection Group: activity_log ─────────────────────────────────────
    match /{path=**}/activity_log/{eventId} {
      allow read: if isSuperAdmin()
        || (isAdmin() && resource.data.institutionId == myInstitutionId());
    }

    // ── Collection Group: audit_log ────────────────────────────────────────
    match /{path=**}/audit_log/{eventId} {
      allow read: if isSuperAdmin();
    }

    // ── Deny everything else ───────────────────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
```
